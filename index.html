المستوى الأول 🟢 الحركة الأساسية

📐 استيراد المكتبة:

import turtle # إنشاء سلحفاة t = turtle.Turtle() # إبقاء النافذة مفتوحة turtle.done()

🐢 أوامر الحركة الأساسية

الأمر المعنى مثال
forward(n) امشي للأمام n خطوة t.forward(100)
backward(n) امشي للخلف n خطوة t.backward(50)
right(deg) استدر يميناً deg درجة t.right(90)
left(deg) استدر يساراً deg درجة t.left(45)
penup() ارفع القلم (لا ترسم) t.penup()
pendown() أنزل القلم (ابدأ الرسم) t.pendown()

🐍 مثال 1: رسم خط بسيط

import turtle t = turtle.Turtle() # ارسم خطاً طوله 100 خطوة t.forward(100) turtle.done()

🐍 مثال 2: رسم مربع

import turtle t = turtle.Turtle() # مربع = 4 أضلاع + 4 استدارات 90 درجة t.forward(100) t.right(90) t.forward(100) t.right(90) t.forward(100) t.right(90) t.forward(100) t.right(90) turtle.done()

💡 ملاحظة مهمة:

المربع له 4 أضلاع متساوية، وزاويته 90 درجة!

🌟 قاعدة الزوايا:

لرسم شكل منتظم بـ n أضلاع:

الزاوية = 360 ÷ n

  • مربع (4): 360 ÷ 4 = 90°
  • مثلث (3): 360 ÷ 3 = 120°
  • خماسي (5): 360 ÷ 5 = 72°
  • سداسي (6): 360 6 = 60°

المستوى الثالث 🔴 الدوال والأشكال المتقدمة

🎯 دالة رسم مربع

import turtle t = turtle.Turtle() def draw_square(size, color): t.color(color) for i in range(4): t.forward(size) t.right(90) # ارسم 3 مربعات بألوان مختلفة draw_square(100, "red") t.penup() t.forward(150) t.pendown() draw_square(80, "blue") turtle.done()

مثال 5: رسم نجمة

import turtle t = turtle.Turtle() t.color("gold") t.speed(5) # نجمة = 5 نقاط، زاوية 144 درجة for i in range(5): t.forward(100) t.right(144) turtle.done()

مثال 6: رسم دائرة

import turtle t = turtle.Turtle() t.color("blue") # رسم دائرة نصف قطرها 50 t.circle(50) turtle.done()
🎮 جربها بنفسك - أشكال متعددة
# محاكاة رسم أشكال متعددة print("🎨 معرض الأشكال الهندسية") print("=" * 40) shapes = [ ("مربع", 4, 90, "red"), ("مثلث", 3, 120, "green"), ("خماسي", 5, 72, "blue"), ("سداسي", 6, 60, "purple"), ("نجمة", 5, 144, "gold") ] for name, sides, angle, color in shapes: print(f"\n{name} {color}:") print(f" الأضلاع: {sides}") print(f" الزاوية: {angle}°") for i in range(sides): print(f" خطوة {i+1}: forward(100), right({angle})") print("\n" + "=" * 40) print("✅ تم رسم جميع الأشكال!")

import turtle t = turtle.Turtle() t.speed(0) # أسرع سرعة turtle.bgcolor("black") colors = ["red", "orange", "yellow", "green", "blue", "purple"] for i in range(100): t.color(colors[i % 6]) # تغيير اللون t.forward(i * 2) # زيادة الطول t.right(91) # زاوية غير منتظمة turtle.done()

🎨 المشروع: معرض الرسومات الفنية

هنعمل برنامج يرسم مجموعة أشكال فنية جميلة!

🖼️ معرض الأشكال

🎮 جرب المشروع
# ═══════════════════════════════════════ # معرض الرسومات الفنية # ═══════════════════════════════════════ print("🎨 مرحباً بك في معرض الرسومات!") print("=" * 40) # دالة رسم مربع def draw_square(size, color): print(f"\n⬜ رسم مربع {color} بحجم {size}:") for i in range(4): print(f" forward({size}), right(90)") # دالة رسم مثلث def draw_triangle(size, color): print(f"\n🔺 رسم مثلث {color} بحجم {size}:") for i in range(3): print(f" forward({size}), right(120)") # دالة رسم نجمة def draw_star(size, color): print(f"\n⭐ رسم نجمة {color} بحجم {size}:") for i in range(5): print(f" forward({size}), right(144)") # دالة رسم دائرة def draw_circle(radius, color): print(f"\n⭕ رسم دائرة {color} بنصف قطر {radius}:") print(f" circle({radius})") # المعرض shapes = [ ("square", 100, "red"), ("triangle", 80, "green"), ("star", 120, "gold"), ("circle", 60, "blue") ] print("\n🖼️ بدء المعرض...") print("-" * 40) for shape, size, color in shapes: if shape == "square": draw_square(size, color) elif shape == "triangle": draw_triangle(size, color) elif shape == "star": draw_star(size, color) elif shape == "circle": draw_circle(size, color) print("\n" + "=" * 40) print("🎉 انتهى المعرض! شكراً لزيارتك!")
